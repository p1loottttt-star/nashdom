- [ ] `rules.js`:
```js
// Правила записи в дом — те же, что CHECK в supabase/schema.sql (сверяет tests/rules.test.mjs).
export const KIND = /^[a-z]{2,20}$/;
export const ID = /^[A-Za-z0-9_:.-]{1,120}$/;
export const MAX_DATA = 1000000; // байт JSON; в базе — pg_column_size
export function checkItem(kind, id, data) {
  if (!KIND.test(kind)) throw new Error(`bad kind: ${kind}`);
  if (!ID.test(id)) throw new Error(`bad id: ${id}`);
  if (data != null && new Blob([JSON.stringify(data)]).size >= MAX_DATA) throw new Error('too big');
}
```

### Задача 8. CI
**Файлы:** `tests/run.mjs`, `.github/workflows/ci.yml`.
```js
// все tests/*.test.mjs по очереди, первый упавший — код выхода 1
import { readdirSync } from 'node:fs';
import { spawnSync } from 'node:child_process';
let bad = 0;
for (const f of readdirSync(new URL('.', import.meta.url)).filter((f) => f.endsWith('.test.mjs'))) {
  const r = spawnSync(process.execPath, [new URL(f, import.meta.url).pathname.replace(/^\/(\w:)/, '$1')], { stdio: 'inherit' });
  if (r.status) { bad++; console.error('FAIL', f); }
}
process.exit(bad ? 1 : 0);
```
